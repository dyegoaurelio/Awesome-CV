%!TEX TS-program = xelatex
%!TEX encoding = UTF-8 Unicode
%
% Plain article-class config with awesome-cv command compatibility
%

\documentclass[11pt, letterpaper]{article}

\usepackage[utf8]{inputenc}
\usepackage[T1]{fontenc}
\usepackage{fontspec}
\usepackage[margin=0.65in, top=0.40in, bottom=0.40in]{geometry}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{hyperref}
\usepackage{xcolor}

% Font setup - clean serif font
\setmainfont{Liberation Serif}

% Hyperlink styling - no colored boxes
\hypersetup{
  colorlinks=true,
  linkcolor=black,
  urlcolor=black,
  pdfborderstyle={/S/U/W 0}
}

% Section formatting - bold with thin rule below
\titleformat{\section}
  {\normalsize\bfseries}
  {}
  {0pt}
  {\MakeUppercase}
  [\vspace{-6pt}\rule{\textwidth}{0.4pt}]
\titlespacing{\section}{0pt}{5pt}{2pt}

% No paragraph indentation, tight paragraph spacing, ragged right (no justification)
\setlength{\parindent}{0pt}
\setlength{\parskip}{2pt}
\raggedright

% Bullet list styling
\setlist[itemize]{
  leftmargin=1.5em,
  label=$\bullet$,
  nosep,
  topsep=2pt,
  itemsep=2pt,
  parsep=0pt,
  font=\small
}
\let\olditemize\itemize
\renewcommand{\itemize}{\small\olditemize}

% Remove page numbers
\pagestyle{empty}

%-------------------------------------------------------------------------------
% Awesome-CV command redefinitions
%-------------------------------------------------------------------------------

% \cvsection{title} -> \section*{title}
\newcommand{\cvsection}[1]{\section*{#1}}

% cventries environment -> no-op wrapper
\newenvironment{cventries}{}{}

% \cventry{pos}{org}{loc}{date}{desc} -> formatted entry
\newcommand{\cventry}[5]{%
  \vspace{3pt}
  \noindent\textbf{#2 \textbar\ #3} \hfill \textbf{#4}\\[0pt]
  \textit{#1}#5%
}

% cvitems environment -> itemize
\newenvironment{cvitems}{\begin{itemize}}{\end{itemize}}

% cvskills environment -> no-op wrapper
\newenvironment{cvskills}{}{}

% \cvskill{category}{skills} -> bold category + skills
\newcommand{\cvskill}[2]{%
  \textbf{#1:} #2\\[2pt]
}

% cvparagraph environment -> noindent text
\newenvironment{cvparagraph}{\noindent}{}
